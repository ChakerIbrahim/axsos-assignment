/* -----------------------------------------------------
   1. Create a database called 'my_first_db'
   ----------------------------------------------------- */
use my_first_db


/* -----------------------------------------------------
   2. Create the 'students' collection
   ----------------------------------------------------- */
db.createCollection("students")


/* -----------------------------------------------------
   3. Create 5 students with the appropriate info
   Format: {name, home_state, lucky_number, birthday:{month, day, year}}
   ----------------------------------------------------- */
db.students.insertMany([
  { name: "Zoe Lin",      home_state: "California", lucky_number: 7, birthday: { month: 4,  day: 12, year: 1998 } },
  { name: "Marcus Bell",  home_state: "Washington", lucky_number: 3, birthday: { month: 9,  day: 3,  year: 1995 } },
  { name: "Priya Raman",  home_state: "California", lucky_number: 5, birthday: { month: 1,  day: 27, year: 2000 } },
  { name: "Diego Ortega", home_state: "Washington", lucky_number: 9, birthday: { month: 11, day: 8,  year: 1997 } },
  { name: "Hana Ito",     home_state: "Utah",       lucky_number: 2, birthday: { month: 6,  day: 19, year: 1999 } }
])


/* -----------------------------------------------------
   4. Get all students
   ----------------------------------------------------- */
db.students.find({})


/* -----------------------------------------------------
   5. Retrieve all students from California (San Jose Dojo)
      or Washington (Seattle Dojo)
   ----------------------------------------------------- */
db.students.find({ home_state: { $in: ["California", "Washington"] } })


/* -----------------------------------------------------
   6. Get all students whose lucky number is greater than 3
   ----------------------------------------------------- */
db.students.find({ lucky_number: { $gt: 3 } })


/* -----------------------------------------------------
   7. Add an 'interests' ARRAY to EVERY student in ONE operation
      Contains: 'coding', 'brunch', 'MongoDB'
      (empty filter {} matches all documents)
   ----------------------------------------------------- */
db.students.updateMany(
  {},
  { $set: { interests: ["coding", "brunch", "MongoDB"] } }
)


/* -----------------------------------------------------
   8. Add some unique interests for each particular student
      ($push appends a value to an existing array)
   ----------------------------------------------------- */
db.students.updateOne({ name: "Zoe Lin"      }, { $push: { interests: "rock climbing" } })
db.students.updateOne({ name: "Marcus Bell"  }, { $push: { interests: "jazz guitar"   } })
db.students.updateOne({ name: "Priya Raman"  }, { $push: { interests: "pottery"       } })
db.students.updateOne({ name: "Diego Ortega" }, { $push: { interests: "cycling"       } })
db.students.updateOne({ name: "Hana Ito"     }, { $push: { interests: "baking"        } })


/* -----------------------------------------------------
   9. Add the interest 'taxes' into someone's interest array
   ----------------------------------------------------- */
db.students.updateOne(
  { name: "Hana Ito" },
  { $push: { interests: "taxes" } }
)


/* -----------------------------------------------------
   10. Remove the 'taxes' interest you just added
       ($pull removes a matching value from an array)
   ----------------------------------------------------- */
db.students.updateOne(
  { name: "Hana Ito" },
  { $pull: { interests: "taxes" } }
)


/* -----------------------------------------------------
   11. Remove all students who are from California
   ----------------------------------------------------- */
db.students.deleteMany({ home_state: "California" })


/* -----------------------------------------------------
   12. Remove a student by name
   ----------------------------------------------------- */
db.students.deleteOne({ name: "Hana Ito" })


/* -----------------------------------------------------
   13. Remove a student whose lucky number is greater than 5
       (JUST ONE - deleteOne stops after the first match)
   ----------------------------------------------------- */
db.students.deleteOne({ lucky_number: { $gt: 5 } })


/* -----------------------------------------------------
   14. Add a 'number_of_belts' field to each student, set to 0
   ----------------------------------------------------- */
db.students.updateMany(
  {},
  { $set: { number_of_belts: 0 } }
)


/* -----------------------------------------------------
   15. Increment 'number_of_belts' by 1 for all students
       in Washington (Seattle Dojo)
   ----------------------------------------------------- */
db.students.updateMany(
  { home_state: "Washington" },
  { $inc: { number_of_belts: 1 } }
)


/* -----------------------------------------------------
   16. Add an 'updated_on' field set to the current date
   ----------------------------------------------------- */
db.students.updateMany(
  {},
  { $currentDate: { updated_on: true } }
)


/* -----------------------------------------------------
   Final check - view the remaining documents
   ----------------------------------------------------- */
db.students.find({}).pretty()


/* =====================================================
   NOTE: The delete operations in steps 11-13 are
   cumulative, so only Marcus Bell remains at the end.
   This is the expected result of running the steps
   in the order given.
   ===================================================== */
